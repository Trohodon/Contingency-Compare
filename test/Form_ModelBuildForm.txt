Private Sub btnOutputDir_Click()

    Dim OutputDir As String

    With Application.FileDialog(msoFileDialogFolderPicker)
        .Title = "Please Select an Output Directory for Auto-generated Files"

        If Dir(txtOutputDir.Value, vbDirectory) <> vbNullString Then
            .InitialFileName = txtOutputDir.Value
        Else
            .InitialFileName = "C:"
        End If

        If .Show = -1 Then
            'okclicked
            OutputDir = .SelectedItems(1)
            Me.txtOutputDir.SetFocus
            Me.txtOutputDir.Text = OutputDir
        Else
            'do nothing. User clicked the cancel button
        End If
    End With

End Sub

            DoCmd.SetWarnings False

            SQL = "UPDATE Snapshots SET [model_name] = '" & Right(Year(Now()), 2) & snapSeason & "', [year] = '" & Year(Now()) & "', [date] = '" & month(Now()) & "/" & day(Now()) & "/" & Year(Now()) & _

            DoCmd.SetWarnings True

            BuildTimer.FinishTiming

            counter = counter + 1

    End If ' Multiple Mods If Not Queue

    sprDispDTM = 1572

Private Sub setLoadValue(allowPercSolChange As Boolean)
    Dim snapDate As Date, snapSeason As String, SnapLoad As Integer, SnapType As String, SnapIntOR As String
    Dim percSolarGenMax As Integer
    Dim baseYear As Integer
    Dim SQL As String
    Dim Snapshot As Integer

    baseYear = Year(Now()) - 2 ' Default to 2 years ago to be safe

    ' Me.SnapShotDatePicker.SetFocus
    snapDate = Me.SnapShotDatePicker.Value

    On Error Resume Next

    ' Me.ComboSnapShotSeason.SetFocus
    If IsNull(Me.ComboSnapShotSeason.Value) Then
        Exit Sub
    ElseIf IsNull(Me.ComboSnapShotType.Value) And Not chkProjectsOutput.Value Then
        Exit Sub
    End If

    snapSeason = Me.ComboSnapShotSeason.Value
    SnapType = Me.ComboSnapShotType.Value

    ' If ChkUseSeasonalRatings.Value = False Then 'Added 10/2023 BU
    '     If Right(snapSeason, 1) = "W" Then
    '         Me.chkUseOldWinRatings.Value = True
    '     Else
    '         Me.chkUseOldWinRatings.Value = False
    '     End If
    ' Else 'then use seasonal ratings
    'Me.chkUseWinRatings.Value = False

    If Right(snapSeason, 1) = "W" And SnapType <> "NTSG" Then 'Option 1 is Winter Peak Ratings
        Me.SeasonalRatingsOptGroup.Value = 1

    ElseIf Right(snapSeason, 1) = "W" And SnapType = "NTSG" Then 'Option 2 is Ops Planning Winter Peak Ratings
        Me.SeasonalRatingsOptGroup.Value = 2

    ElseIf Right(snapSeason, 1) = "Z" Or Right(snapSeason, 1) = "L" Or Left(snapSeason, 1) = "D" Then 'Option 3 is Spring Ratings
        Me.SeasonalRatingsOptGroup.Value = 3

    ElseIf Right(snapSeason, 1) = "S" Then 'Option 4 is Summer Ratings
        Me.SeasonalRatingsOptGroup.Value = 4

    ElseIf Right(snapSeason, 1) = "F" Then 'Option 5 is Fall Ratings
        Me.SeasonalRatingsOptGroup.Value = 5

    ElseIf Right(snapSeason, 1) = "H" Then 'Option 4 is Summer Ratings
        Me.SeasonalRatingsOptGroup.Value = 4

    Else
        Me.SeasonalRatingsOptGroup.Value = 2
    End If
    'End If

    If Not chkProjectsOutput.Value Then _
        SnapType = Me.ComboSnapShotType.Value

    If SnapType = "Sh Ckt" Then
        Me.chkPSSESeq.Value = True
    End If

    ' If SnapType = "GEN CNXN" Then
    '     Me.CheckNoIA.Value = True
    ' Else
    '     Me.CheckNoIA.Value = False
    ' End If

    On Error GoTo 0

    If Not chkProjectsOutput.Value Then
        SnapLoad = returnLoadValue(snapDate, snapSeason, SnapType)
        ' Me.SnapShotLoad.SetFocus
        Me.SnapShotLoad.Value = SnapLoad
    End If

    If allowPercSolChange Then
        If Not chkProjectsOutput.Value Then
            percSolarGenMax = returnPercGenMax(snapSeason, SnapType)
        Else
            percSolarGenMax = returnPercGenMax(snapSeason, "")
        End If

        ' Me.TextPercSolarOfPeak.SetFocus
        Me.TextPercSolarOfPeak.Value = percSolarGenMax
    End If

    If Not chkProjectsOutput.Value Then
        baseYear = returnBaseYear(baseYear, Me.CheckLastYearChecked.Value)
        ' Me.txtBaseYear.SetFocus
        Me.txtBaseYear.Value = baseYear
    End If

    ' Set up Snapshots table based on selections in ModelBuildForm

    Dim db As DAO.Database
    Dim Snapshots As DAO.Recordset

    DoCmd.SetWarnings False

    snapDate = Me.SnapShotDatePicker.Value

    If IsNull(Me.ComboSnapShotSeason.Value) Then
        Exit Sub
    End If

    'Set snapshot up
    If IsNull(Me.SnapShotLoad.Value) Then
        Exit Sub
    End If

    SnapLoad = Me.SnapShotLoad.Value
    SnapType = Me.ComboSnapShotType.Value

    If IsNull(Me.ComboSnapShotType.Value) Then
        Exit Sub
    End If

    On Error Resume Next
    SnapIntOR = Me.SnapShotIntOR.Value
    On Error GoTo -1

    Set db = CurrentDb()
    Set Snapshots = db.OpenRecordset("Snapshots", dbOpenDynaset, dbSeeChanges)

    Snapshots.MoveLast
    Snapshots.MoveFirst

    SQL = "UPDATE Snapshots SET [Selected] = '-1'"
    DoCmd.RunSQL SQL

    While Not Snapshots.EOF

        If Snapshots.Fields("Selected") Then

            Snapshot = Snapshots.Fields("model_index")

            SQL = "UPDATE Snapshots SET [model_name] = '" & Right(Year(snapDate), 2) & snapSeason & "', [year] = '" & Year(snapDate) & "', [season] = '" & snapSeason & "', [date] = '" _
                & month(snapDate) & "/" & day(snapDate) & "/" & Year(snapDate) & "', [system_load_MW] = '" & SnapLoad & "', [OR_interchange_MW] = '" & SnapIntOR & "', [study] = '" & SnapType _
                & " " & Year(snapDate) & "', [psse_header1] = '" & Year(snapDate) & " " & SnapType & "', [psse_header2] = '" & Year(snapDate) & snapSeason & " SCEG AREA'"

            DoCmd.RunSQL SQL

            If Right(snapSeason, 1) = "W" And month(snapDate) <> 12 Then
                SQL = "UPDATE Snapshots SET [model_name] = '" & Right(Year(snapDate), 2) - 1 & snapSeason & "', [year] = '" & Year(snapDate) - 1 & "', [study] = '" & SnapType _
                    & " " & Year(snapDate) - 1 & "', [psse_header1] = '" & Year(snapDate) - 1 & " " & SnapType & "', [psse_header2] = '" & Year(snapDate) - 1 & snapSeason & " SCEG AREA'"

                DoCmd.RunSQL SQL
            End If

        Else
            Snapshots.Delete
        End If

        Snapshots.MoveNext

    Wend

    Snapshots.Close

End Sub

Private Sub BtnAddQueue_Click() 'Adds Item to Model Build Queue

    If Me.chkGenDown.Value Or Me.CheckContFacStudy.Value Then
        MsgBox ("Gen Down and Contingent Facilities Studies are currently not supported for Multiple Model Queue.")
        Exit Sub
    End If

    VarLoadFail = False
    Dim ListSuffix As String

    If Me.chkPSSESav.Value Then
        If MsgBox("You have selected to write out a PSSE Python Script. Do you wish to continue?", vbYesNo, "Output Selection") = vbNo Then
            Exit Sub
        End If
        ListSuffix = ".py"

    ElseIf Me.chkPWaux.Value Then
        If MsgBox("You have selected to write out a PowerWorld .aux file. Do you wish to continue?", vbYesNo, "Output Selection") = vbNo Then
            Exit Sub
        End If
        ListSuffix = ".aux"
    End If

    QueueCount = ModelQueue.ListCount

    If QueueCount < 10 Then

        SetQueuedVariables (QueueCount)

        If VarLoadFail Then 
            Exit Sub
        End If

        If Me.ComboSnapShotSeason.Value = "W" And month(Me.SnapShotDatePicker) <> 12 Then
            ModelQueue.AddItem Item:=Right(Year(Me.SnapShotDatePicker.Value) - 1, 2) & Me.ComboSnapShotSeason.Value & ListSuffix
        Else
            ModelQueue.AddItem Item:=Right(Year(Me.SnapShotDatePicker.Value), 2) & Me.ComboSnapShotSeason.Value & ListSuffix
        End If

    Else
        MsgBox ("Queue is limited to 10 entries.")
    End If

    UseQueue = True
    Me.Refresh

End Sub
